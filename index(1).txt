    function renderColorPalette() {
      const palette = document.getElementById('colorPalette');
      palette.innerHTML = predefinedColors.map(color => `
        <div class="color-option" style="background: ${color};" onclick="toggleColorSelection('${color}')"></div>

    function toggleColorSelection(color) {
      const index = selectedCustomColors.indexOf(color);
      if (index > -1) {
        selectedCustomColors.splice(index, 1);
      } else {
        if (selectedCustomColors.length < 5) {
          selectedCustomColors.push(color);
        } else {
          showToast('Max 5 colors allowed', 'warning');
          return;
        }
      }
      updateColorSelection();
    }




    function updateColorSelection() {
      const colors = document.querySelectorAll('.color-option');
      colors.forEach(el => {
        el.classList.remove('selected');
      });
      
      selectedCustomColors.forEach(color => {
        document.querySelector(`[onclick*="${color}"]`).classList.add('selected');
      });




      const preview = document.getElementById('selectedColorsPreview');
      if (selectedCustomColors.length > 0) {
        const gradient = `linear-gradient(90deg, ${selectedCustomColors.join(', ')})`;
        preview.style.background = gradient;
        preview.textContent = '';
        preview.classList.remove('empty');
      } else {
        preview.classList.add('empty');
        preview.textContent = 'No colors selected';
      }
    }

let isLoadingDates = false;  // Add this at the top with other variables


function loadDatesInBackground() {
  // Prevent multiple simultaneous calls
  if (isLoadingDates) return;
  
  isLoadingDates = true;
  const fileLinks = portalData.allFiles.map(f => f.link);
  
  google.script.run.withSuccessHandler(function(dates) {
    isLoadingDates = false;  // Reset flag when done
    
    portalData.allFiles.forEach(file => {
      if (dates[file.link]) {
        file.lastUpdated = dates[file.link];
      }
    });


    portalData.frequentFiles.forEach(file => {
      if (dates[file.link]) {
        file.lastUpdated = dates[file.link];
      }
    });


    // Re-render to show updated dates
    if (currentView === 'frequent') {
      renderFrequentFiles();
    } else if (currentView === 'all') {
      renderAllFiles();
    }
  }).withFailureHandler(function() {
    isLoadingDates = false;  // Reset on failure too
  }).getFilesDatesAsync(fileLinks);
}




    function applyThemeToPage(theme) {
      document.body.classList.remove('bw-theme', 'custom-gradient');
      
      if (theme === 'bw') {
        document.body.classList.add('bw-theme');
      } else if (theme.startsWith('custom:')) {
        const colors = theme.split(':').slice(1);
        const gradient = `linear-gradient(135deg, ${colors.join(', ')})`;
        document.body.style.background = gradient;
        document.body.classList.add('custom-gradient');
      }
    }

    }

    function renderAllFiles() {
      const container = document.getElementById('allFilesList');
      let files = portalData.allFiles;




      if (files.length === 0) {
        container.innerHTML = `
          <div style="grid-column: 1/-1;">
            <div class="empty-state">
              <div class="empty-state-icon"><i class="fas fa-folder-open"></i></div>
              <div class="empty-state-text">No files available.</div>
            </div>
          </div>
        `;
        return;
      }
      
      const grouped = {};
      files.forEach(file => {
        if (!grouped[file.category]) {
          grouped[file.category] = [];
        }
        grouped[file.category].push(file);
      });
      
      // Sort files within each category by lastUpdated (newest first)
      Object.keys(grouped).forEach(category => {
        grouped[category].sort((a, b) => {
          const dateA = new Date(a.lastUpdated);
          const dateB = new Date(b.lastUpdated);
          return dateB - dateA;
        });
      });
      
      let html = '';
      
      Object.keys(grouped).sort().forEach(category => {
        html += `
          <div class="file-tile" style="border-left: 4px solid var(--primary);">
            <div class="file-name"><i class="fas fa-folder"></i> ${category}</div>
            <div style="font-size: 12px; color: #999;">${grouped[category].length} file(s)</div>
            <div style="margin-top: 15px;">
              <button class="file-btn file-btn-view" onclick="expandCategory('${escapeQuotes(category)}')">
                <i class="fas fa-chevron-right"></i> Open Folder
              </button>
            </div>
          </div>
        `;
      });
      
      container.innerHTML = html;
    }




    function expandCategory(category) {

      // Sort by lastUpdated (newest first)
      files.sort((a, b) => {
        const dateA = new Date(a.lastUpdated);
        const dateB = new Date(b.lastUpdated);
        return dateB - dateA;
      });




      const container = document.getElementById('allFilesList');
      
      let html = `
        <div style="grid-column: 1/-1; margin-bottom: 20px;">
          <button class="form-btn form-btn-secondary" onclick="renderAllFiles()" style="width: auto; margin-bottom: 15px;">

    function updatePushSelection() {
      const checkboxes = document.querySelectorAll('#pushFilesList input[type="checkbox"]');

        checked.length === checkboxes.length && checkboxes.length > 0;

    function pushFilesToFrequent() {
      if (selectedPushFiles.length === 0) {
        showToast('Please select at least one file', 'error');
        return;
      }
      
      showToastWithSpinner(`Pushing ${selectedPushFiles.length} file(s)...`);
      
      let processed = 0;
      selectedPushFiles.forEach((file, idx) => {
        google.script.run.withSuccessHandler(function(result) {
          processed++;
          if (processed === selectedPushFiles.length) {
            hideToast();
            showToast('Files pushed to Frequent Files', 'success');
            document.getElementById('selectAllCheckbox').checked = false;
            selectedPushFiles = [];
            
            google.script.run.withSuccessHandler(function(data) {
              portalData = data;
              renderDevView();
              renderFrequentFiles();
            }).getPortalDataOptimized();
          }
        }).toggleFrequentFile(file.id, true);
      });
    }

    function searchAllFiles() {
      const query = document.getElementById('allFilesSearch').value.toLowerCase();
      const suggestionsContainer = document.getElementById('allFilesSuggestions');
      
      if (!query) {
        const categories = portalData.categories;
        suggestionsContainer.innerHTML = categories.map(cat => `
          <div class="suggestion-item" onclick="selectAndExpandCategory('${escapeQuotes(cat)}')">
            <i class="fas fa-folder suggestion-icon"></i>
            ${cat}
          </div>
        `).join('');
        suggestionsContainer.classList.add('show');
        return;
      }
      
      const results = [];
      
      portalData.allFiles.forEach(file => {
        if (file.name.toLowerCase().includes(query)) {
          results.push({ type: 'file', name: file.name, category: file.category, id: file.id, link: file.link, embeddedURL: file.embeddedURL });
        }
      });
      
      if (results.length === 0) {
        suggestionsContainer.innerHTML = '<div class="suggestion-item">No results found</div>';
      } else {
        suggestionsContainer.innerHTML = results.slice(0, 10).map(result => `
          <div class="suggestion-item" onclick="handleSearchSelect('${result.type}', '${escapeQuotes(result.name)}', ${result.id}, '${escapeQuotes(result.link)}', '${escapeQuotes(result.embeddedURL)}')">
            <i class="fas fa-file suggestion-icon"></i>
            ${result.name}
            <span class="category-badge">${result.category}</span>
          </div>
        `).join('');
      }
      
      suggestionsContainer.classList.add('show');
    }

    function openThemeSelector() {
      selectedTheme = currentTheme;
      selectedCustomColors = [];
      
      // Parse custom colors from current theme if it's custom
      if (selectedTheme.startsWith('custom:')) {
        selectedCustomColors = selectedTheme.split(':').slice(1);
      }
      
      document.getElementById('themeSelector').classList.add('show');
      
      // Show reset button if current theme is custom
      const resetBtn = document.getElementById('resetThemeBtn');
      if (selectedTheme.startsWith('custom:')) {
        resetBtn.style.display = 'block';
      } else {
        resetBtn.style.display = 'none';
      }
      
      // Update color selection display if needed
      if (selectedTheme === 'custom' || selectedTheme.startsWith('custom:')) {
        document.getElementById('colorPaletteSection').classList.remove('hidden');
        updateColorSelection();
      } else {
        document.getElementById('colorPaletteSection').classList.add('hidden');
      }
      
      // Highlight selected theme card
      document.querySelectorAll('.theme-card').forEach(card => {
        if (card.dataset.theme === selectedTheme || 
            (selectedTheme.startsWith('custom:') && card.dataset.theme === 'custom')) {
          card.classList.add('selected');
        } else {
          card.classList.remove('selected');
        }
      });
    }




    function closeThemeSelector() {
      document.getElementById('themeSelector').classList.remove('show');
      selectedCustomColors = [];
      updateColorSelection();
    }




    function selectTheme(theme) {
      selectedTheme = theme;
      selectedCustomColors = [];
      
      document.querySelectorAll('.theme-card').forEach(card => {
        card.classList.remove('selected');
      });
      event.target.closest('.theme-card').classList.add('selected');
      
      if (theme === 'custom') {
        document.getElementById('colorPaletteSection').classList.remove('hidden');
      } else {
        document.getElementById('colorPaletteSection').classList.add('hidden');
      }
    }




    function resetToDefault() {
      selectedTheme = 'default';
      selectedCustomColors = [];
      
      document.querySelectorAll('.theme-card').forEach(card => {
        card.classList.remove('selected');
      });
      document.querySelector('[data-theme="default"]').classList.add('selected');
      
      document.getElementById('colorPaletteSection').classList.add('hidden');
      updateColorSelection();
    }




    function applyTheme() {
      let themeToSave = selectedTheme;
      
      if (selectedTheme === 'custom') {
        if (selectedCustomColors.length < 2) {
          showToast('Please select at least 2 colors', 'warning');
          return;
        }
        themeToSave = 'custom:' + selectedCustomColors.join(':');
      }
      
      currentTheme = themeToSave;
      applyThemeToPage(currentTheme);
      
      google.script.run.withSuccessHandler(function(result) {
        if (result.success) {
          closeThemeSelector();
          showToast('Theme applied successfully', 'success');
        }
      }).saveUserTheme(portalData.userEmail, currentTheme);
    }




    function showToast(message, type = 'info') {
      hideToast();
      const toast = document.createElement('div');
      toast.className = `toast ${type}`;
      toast.id = 'activeToast';
      toast.innerHTML = `<div class="toast-spinner"></div><div class="toast-message">${message}</div>`;
      document.body.appendChild(toast);
      
      if (type !== 'loading') {

        }, 3000);
      }
    }
